model ReassignedFlag
  input Boolean useJoint;
  input Boolean reset;
  output Boolean flag(start=false, fixed=true);
algorithm
  when sample(0.0, 0.01) then
    if not useJoint then
      flag := true;
    else
      flag := not reset and pre(flag);
    end if;
    if useJoint then
      flag := true;
    end if;
  end when;
end ReassignedFlag;
