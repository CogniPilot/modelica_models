model ReassignedFlagParameterStatement
  parameter Boolean useJoint=false;
  input Boolean reset;
  input Boolean nextFlag;
  output Boolean flag(start=false, fixed=true);
protected
  Boolean auxiliaryFlag;
algorithm
  when sample(0.0, 0.01) then
    if not useJoint then
      auxiliaryFlag := true;
    else
      auxiliaryFlag := not reset and pre(flag);
    end if;
    if useJoint then
      flag := nextFlag;
    else
      flag := auxiliaryFlag;
    end if;
  end when;
end ReassignedFlagParameterStatement;
