model ReassignedFlagFixed
  input Boolean useJoint;
  input Boolean reset;
  output Boolean flag(start=false, fixed=true);
protected
  Boolean auxiliaryFlag;
algorithm
  when sample(0.0, 0.01) then
    if not useJoint then
      auxiliaryFlag := true;
    else
      auxiliaryFlag := not reset and pre(flag);
    end if;
    flag := if useJoint then true else auxiliaryFlag;
  end when;
end ReassignedFlagFixed;
